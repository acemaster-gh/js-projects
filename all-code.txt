const themeBtn = document.querySelector('#themeBtn');
const input = document.querySelector('#guess-input');
const submit = document.querySelector('#submitBtn');
const reset = document.querySelector('#resetBtn');
const feedbackMessage = document.querySelector('#message');
const attempts = document.querySelector('#attempts');
const status = document.querySelector('#status');
const history = document.querySelector('#history');






//EVENT LISTENERS


themeBtn.addEventListener('click', () => {
    document.body.classList.toggle('dark-mode');

    if (document.body.classList.contains('dark-mode')) {
        localStorage.setItem('theme', 'dark');
        themeBtn.textContent = 'Switch to Light Mode';

    } else {
        localStorage.setItem('theme', 'light');
        themeBtn.textContent = 'Switch to Dark Mode';
    }

});

// step 1 - STATE OBJECT

const state = {
  min: 1,
  max: 100,
  secretNumber: null,
  attempts: 0,
  maxAttempts: 10,
  status: "idle", 
  history: []
};
// step 2 - GENERATING SECRET NUMBER (INITIALISATION)

function generateSecretNumber(min, max) {
    return Math.floor(Math.random( ) * (max - min + 1)) + min;
}

// state.secretNumber = generateSecretNumber(state.min, state.max);

// step 3 starting game

function startGame(){
    state.secretNumber = generateSecretNumber(state.min, state.max);
    state.attempts = 0;
    state.history = [];
    state.status = "playing";
    attempts.textContent = state.attempts;
    feedbackMessage.textContent = "Game started! Make a guess.";
    input.disabled = false;
    input.value = "";
}

// step 4 VALIDATION

function validateGuess(value) {
  if (state.status !== "playing") {
    return { valid: false, message: "Game is over. Reset to play again." };
  }

  if (value.trim() === "") {
    return { valid: false, message: "Input cannot be empty." };
  }

  const number = Number(value);

  if (isNaN(number)) {
    return { valid: false, message: "Please enter a valid number." };
  }

  if (!Number.isInteger(number)) {
    return { valid: false, message: "No decimals allowed." };
  }

  if (number < state.min || number > state.max) {
    return { valid: false, message: `Enter a number between ${state.min} and ${state.max}.` };
  }

  return { valid: true, number };
}

// step 5 HANDLE GUESS

function handleGuess(guess) {
  state.attempts++;
  state.history.push(guess);
  attempts.textContent = state.attempts;

  if (guess === state.secretNumber) {
    state.status = "won";
    input.disabled = true;
    return "Correct! You won!";
  }

  if (state.attempts >= state.maxAttempts) {
    state.status = "lost";
    input.disabled = true;
    return `Game over! The number was ${state.secretNumber}`;
  }

  if (guess > state.secretNumber) {
    return "Too high!";
  } else {
    return "Too low!";
  }
}


// STEP 6 CONNECTING THE SUBMIT BUTTON

submit.addEventListener("click", () => {
  const value = input.value;
  const validation = validateGuess(value);

  if (!validation.valid) {
    feedbackMessage.textContent = validation.message;
    return;
  }

  const resultMessage = handleGuess(validation.number);
  feedbackMessage.textContent = resultMessage;
});

reset.addEventListener('click', () => {
    startGame();
});

startGame();

localStorage.setItem("theme","dark")

themeIcon.classList.remove("bx-moon")
themeIcon.classList.add("bx-sun")

}else{

themeIcon.classList.remove("bx-sun")
themeIcon.classList.add("bx-moon")

}

})

/* RESTORE THEME */

const savedTheme = localStorage.getItem("theme")

if(savedTheme === "dark"){

document.body.classList.add("dark-mode")

themeIcon.classList.remove("bx-moon")
themeIcon.classList.add("bx-sun")

}

    <script>
        /* ============================================================
           JAVASCRIPT — Organised in clear sections
           Read this top to bottom like a story.
           ============================================================ */


        /* ============================================================
           1. CONFIGURATION
           — All magic numbers live here. Never scatter them in code.
           ============================================================ */
        const CONFIG = {
            modes: {
                work: { label: 'focus session', minutes: 25 },
                break: { label: 'short break', minutes: 5 },
                long: { label: 'long break', minutes: 15 },
            },
            longBreakAfter: 4,   // sessions before a long break
            ringCircumference: 628, // 2 * Math.PI * 100 (matches CSS)
        };


        /* ============================================================
           2. STATE
           — One object holds ALL app state. Makes debugging easy.
           — Never spread state randomly across variables.
           ============================================================ */
        const state = {
            mode: 'work',
            isRunning: false,
            totalSeconds: CONFIG.modes.work.minutes * 60,
            remainingSeconds: CONFIG.modes.work.minutes * 60,
            intervalId: null,
            completedSessions: 0,
            minutesFocused: 0,
            streak: 0,
            sessionHistory: [],   // array of 'work' | 'break' | 'long'
            settings: {
                work: 25,
                break: 5,
                long: 15,
                sound: true,
                focusMode: false,
            },
        };


        /* ============================================================
           3. DOM REFERENCES
           — Grab all elements once at the top.
           — Never call getElementById inside event handlers or loops.
           ============================================================ */
        const DOM = {
            timerDisplay: document.getElementById('timer-display'),
            timerLabel: document.getElementById('timer-label'),
            ring: document.getElementById('ring'),
            btnPlay: document.getElementById('btn-play'),
            btnReset: document.getElementById('btn-reset'),
            btnSkip: document.getElementById('btn-skip'),
            modeTabs: document.querySelectorAll('.mode-tab'),
            statSessions: document.getElementById('stat-sessions'),
            statMinutes: document.getElementById('stat-minutes'),
            statStreak: document.getElementById('stat-streak'),
            sessionDots: document.getElementById('session-dots'),
            settingsBtn: document.getElementById('settings-btn'),
            settingsPanel: document.getElementById('settings-panel'),
            stepperBtns: document.querySelectorAll('.stepper__btn'),
            settingWork: document.getElementById('setting-work'),
            settingBreak: document.getElementById('setting-break'),
            settingLong: document.getElementById('setting-long'),
            toggleSound: document.getElementById('toggle-sound'),
            toggleFocus: document.getElementById('toggle-focus'),
            toast: document.getElementById('toast'),
        };


        /* ============================================================
           4. UTILITY FUNCTIONS
           — Small, reusable, pure functions. Do one thing each.
           ============================================================ */

        /** Format seconds into MM:SS string */
        function formatTime(seconds) {
            const m = Math.floor(seconds / 60).toString().padStart(2, '0');
            const s = (seconds % 60).toString().padStart(2, '0');
            return `${m}:${s}`;
        }

        /** Calculate ring progress (0 = full, circumference = empty) */
        function calcRingOffset(remaining, total) {
            const progress = remaining / total;
            return CONFIG.ringCircumference * (1 - progress);
        }

        /** Show a toast message for a short duration */
        function showToast(message, duration = 2800) {
            DOM.toast.textContent = message;
            DOM.toast.classList.add('show');
            setTimeout(() => DOM.toast.classList.remove('show'), duration);
        }

        /** Play a soft beep using the Web Audio API (no audio file needed) */
        function playBeep(type = 'end') {
            if (!state.settings.sound) return;
            try {
                const ctx = new AudioContext();
                const osc = ctx.createOscillator();
                const gain = ctx.createGain();
                osc.connect(gain);
                gain.connect(ctx.destination);

                osc.type = 'sine';
                osc.frequency.value = type === 'end' ? 528 : 440;
                gain.gain.setValueAtTime(0.3, ctx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 1);

                osc.start(ctx.currentTime);
                osc.stop(ctx.currentTime + 1);
            } catch (e) {
                // AudioContext blocked (no user interaction yet) — fail silently
            }
        }

        /** Update the browser tab title */
        function updatePageTitle() {
            document.title = state.isRunning
                ? `${formatTime(state.remainingSeconds)} — TEMPO`
                : 'TEMPO — Focus Timer';
        }


        /* ============================================================
           5. RENDER FUNCTIONS
           — These READ from state and UPDATE the DOM.
           — Never update the DOM directly in logic functions.
           ============================================================ */

        function renderTimer() {
            DOM.timerDisplay.textContent = formatTime(state.remainingSeconds);
            DOM.ring.style.strokeDashoffset = calcRingOffset(state.remainingSeconds, state.totalSeconds);
            updatePageTitle();
        }

        function renderControls() {
            DOM.btnPlay.innerHTML = state.isRunning
                ? '⏸ &nbsp;Pause'
                : '▶ &nbsp;' + (state.remainingSeconds < state.totalSeconds ? 'Resume' : 'Start');
        }

        function renderModeUI() {
            document.body.setAttribute('data-mode', state.mode);
            DOM.timerLabel.textContent = CONFIG.modes[state.mode].label;

            DOM.modeTabs.forEach(tab => {
                const isActive = tab.dataset.mode === state.mode;
                tab.classList.toggle('active', isActive);
                tab.setAttribute('aria-selected', isActive);
            });
        }

        function renderStats() {
            DOM.statSessions.textContent = state.completedSessions;
            DOM.statMinutes.textContent = state.minutesFocused;
            DOM.statStreak.textContent = state.streak;
        }

        function renderSessionDots() {
            DOM.sessionDots.innerHTML = '';
            state.sessionHistory.slice(-12).forEach(type => {
                const dot = document.createElement('div');
                dot.className = `dot ${type}`;
                dot.setAttribute('role', 'listitem');
                dot.setAttribute('aria-label', `${type} session completed`);
                DOM.sessionDots.appendChild(dot);
            });
        }

        /** Full re-render — call this any time state changes */
        function render() {
            renderTimer();
            renderControls();
            renderModeUI();
            renderStats();
            renderSessionDots();
        }


        /* ============================================================
           6. CORE TIMER LOGIC
           ============================================================ */

        function startTimer() {
            if (state.isRunning) return;
            state.isRunning = true;

            state.intervalId = setInterval(() => {
                state.remainingSeconds--;

                if (state.remainingSeconds <= 0) {
                    onSessionComplete();
                } else {
                    renderTimer();
                    updatePageTitle();
                }
            }, 1000);

            renderControls();
        }

        function pauseTimer() {
            if (!state.isRunning) return;
            state.isRunning = false;
            clearInterval(state.intervalId);
            renderControls();
        }

        function resetTimer() {
            pauseTimer();
            state.remainingSeconds = state.totalSeconds;
            render();
        }

        function onSessionComplete() {
            pauseTimer();
            playBeep('end');

            // Record history
            state.sessionHistory.push(state.mode);

            if (state.mode === 'work') {
                state.completedSessions++;
                state.minutesFocused += state.settings.work;
                state.streak++;

                const isLongBreakTime = state.completedSessions % CONFIG.longBreakAfter === 0;
                showToast(isLongBreakTime
                    ? `🎉 ${CONFIG.longBreakAfter} sessions done! Take a long break.`
                    : '✓ Session complete! Time for a short break.');

                // Auto-switch mode
                switchMode(isLongBreakTime ? 'long' : 'break');
            } else {
                showToast('Break over — back to focus!');
                switchMode('work');
            }

            renderStats();
            renderSessionDots();
        }

        function skipSession() {
            pauseTimer();
            state.sessionHistory.push(state.mode);
            switchMode(state.mode === 'work'
                ? (state.completedSessions % CONFIG.longBreakAfter === 0 ? 'long' : 'break')
                : 'work');
        }

        function switchMode(newMode) {
            state.mode = newMode;
            state.totalSeconds = state.settings[newMode] * 60;
            state.remainingSeconds = state.totalSeconds;
            state.isRunning = false;
            clearInterval(state.intervalId);
            render();
        }


        /* ============================================================
           7. SETTINGS LOGIC
           ============================================================ */

        function adjustSetting(target, direction) {
            const limits = { work: [5, 90], break: [1, 30], long: [5, 60] };
            const [min, max] = limits[target];

            state.settings[target] = Math.min(max, Math.max(min,
                state.settings[target] + direction
            ));

            // Update display
            const displayEl = document.getElementById(`setting-${target}`);
            if (displayEl) displayEl.textContent = state.settings[target];

            // Update matching mode tab data
            const tab = document.querySelector(`.mode-tab[data-mode="${target}"]`);
            if (tab) tab.dataset.minutes = state.settings[target];

            // If we're currently in this mode, update the timer too
            if (state.mode === target && !state.isRunning) {
                state.totalSeconds = state.settings[target] * 60;
                state.remainingSeconds = state.totalSeconds;
                renderTimer();
            }
        }


        /* ============================================================
           8. EVENT LISTENERS
           — Grouped at the bottom, clearly separated.
           — Each listener does ONE thing or calls a named function.
           ============================================================ */

        // Play / Pause
        DOM.btnPlay.addEventListener('click', () => {
            state.isRunning ? pauseTimer() : startTimer();
        });

        // Reset
        DOM.btnReset.addEventListener('click', resetTimer);

        // Skip
        DOM.btnSkip.addEventListener('click', skipSession);

        // Mode tabs
        DOM.modeTabs.forEach(tab => {
            tab.addEventListener('click', () => {
                const newMode = tab.dataset.mode;
                if (newMode !== state.mode) switchMode(newMode);
            });
        });

        // Settings panel toggle
        DOM.settingsBtn.addEventListener('click', () => {
            const isOpen = DOM.settingsPanel.classList.toggle('open');
            DOM.settingsBtn.setAttribute('aria-expanded', isOpen);
            DOM.settingsBtn.textContent = isOpen ? '✕ Close' : '⚙ Settings';
        });

        // Stepper buttons (settings)
        DOM.stepperBtns.forEach(btn => {
            btn.addEventListener('click', () => {
                adjustSetting(btn.dataset.target, parseInt(btn.dataset.dir));
            });
        });

        // Sound toggle
        DOM.toggleSound.addEventListener('change', () => {
            state.settings.sound = DOM.toggleSound.checked;
        });

        // Focus mode toggle
        DOM.toggleFocus.addEventListener('change', () => {
            state.settings.focusMode = DOM.toggleFocus.checked;
            document.body.classList.toggle('focus-mode', state.settings.focusMode);
            showToast(state.settings.focusMode ? 'Focus mode on — distractions hidden' : 'Focus mode off');
        });

        // Keyboard shortcuts
        document.addEventListener('keydown', (e) => {
            // Don't fire if user is typing in an input
            if (e.target.tagName === 'INPUT') return;

            switch (e.code) {
                case 'Space':
                    e.preventDefault();
                    state.isRunning ? pauseTimer() : startTimer();
                    break;
                case 'KeyR':
                    resetTimer();
                    break;
                case 'KeyS':
                    skipSession();
                    break;
            }
        });


        /* ============================================================
           9. INITIALISE
           — Run once on page load to set the initial state.
           ============================================================ */
        render();

    </script>
</body>
